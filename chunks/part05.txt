function shuffle(arr, rng = Math.random) {
  const a = [...arr];
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(rng() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

function buildBoard(deckKey, nPairs, rng = Math.random) {
  const source = deckKey === "mixed"
    ? DECK_KEYS.flatMap((k) => DECKS[k].pairs)
    : DECKS[deckKey]?.pairs;
  if (!source) throw new Error(`Unknown deck: ${deckKey}`);
  if (nPairs > source.length) throw new Error(`Not enough pairs in ${deckKey}`);

  const chosen = shuffle(source, rng).slice(0, nPairs);
  const cards = [];
  chosen.forEach(([q, a], pairId) => {
    const pearl = PEARLS[q] || "";
    cards.push({ id: `${pairId}-q`, pairId, side: "q", text: q, mate: a, pearl });
    cards.push({ id: `${pairId}-a`, pairId, side: "a", text: a, mate: q, pearl });
  });
  return shuffle(cards, rng);
}

function getPairText(board, flipped) {
  const [i, j] = flipped;
  if (i == null || j == null) return { q: "", a: "", pearl: "" };
  const first = board[i];
  const second = board[j];
  const qCard = first.side === "q" ? first : second;
  const aCard = first.side === "a" ? first : second;
  return { q: qCard.text, a: aCard.text, pearl: qCard.pearl || aCard.pearl || "" };
}

function roomCanBeJoined(room) {
  return !!room && room.status === "waiting" && !room.players?.p2;
}
