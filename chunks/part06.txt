function useAdvanceKey(active, fn) {
  useEffect(() => {
    if (!active) return;
    const h = (e) => {
      if (e.key === " " || e.key === "Enter") {
        e.preventDefault();
        fn();
      }
    };
    window.addEventListener("keydown", h);
    return () => window.removeEventListener("keydown", h);
  }, [active, fn]);
}

  const [name, setName] = useState("");

  const [n2, setN2] = useState("");

  const accent = hue(deck);
  const title = { solo: "Solo", duel: "Duel", online: "Play with friends" }[mode];

  return (
    <section className="screen">

      <h2>{title}</h2>
      <div className="eyebrow">Subject</div>

          </button>
        ))}

          <p className="hint">Connect a pair and you keep the turn. Miss and the phone changes hands.</p>
        </div>

        </div>
      )}
      {notice && <div className="notice" style={{ color: accent }}>{notice}</div>}
    </section>
  );
}

function Card({ card, open, matched, shaking, onClick, disabled, accent, cols, pending }) {

      <span className="card-inner">

        <span className={`card-face ${card.side === "q" ? "question" : "answer"}`} style={{ "--accent": accent }}>{card.text}</span>
      </span>
    </button>
  );
}

function Connection({ from, to, cols, rows, accent }) {
  const pt = (i) => ({ x: ((i % cols) + 0.5) * (100 / cols), y: (Math.floor(i / cols) + 0.5) * (100 / rows) });
  const a = pt(from), b = pt(to);
  return (
    <svg className="connection" viewBox="0 0 100 100" preserveAspectRatio="none" aria-hidden="true">
      <line className="connection-glow" x1={a.x} y1={a.y} x2={b.x} y2={b.y} stroke={accent} />
      <line className="connection-line" x1={a.x} y1={a.y} x2={b.x} y2={b.y} stroke={accent} />
      <circle cx={a.x} cy={a.y} r="3" fill={accent} /><circle cx={b.x} cy={b.y} r="3" fill={accent} />
    </svg>
  );
}
